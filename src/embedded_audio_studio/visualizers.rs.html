7pub fn render_visualizers(ui: &mut egui::Ui, vis: &VisualizerData, project: &DawProject) {

70                ui.add_space(6.0);

95fn render_scope(ui: &mut egui::Ui, samples: &[f32], stroke_color: Color32) {
96    let (response, painter) =
97        ui.allocate_painter(Vec2::new(ui.available_width(), 85.0), egui::Sense::hover());
98    let rect = response.rect;
99    painter.rect_filled(rect, 4.0, Color32::from_rgb(12, 15, 22));
100    painter.rect_stroke(
101        rect,
102        4.0,
103        Stroke::new(1.0_f32, Color32::from_rgb(35, 42, 60)),
104        StrokeKind::Inside,
105    );
106
107    let mid_y = rect.center().y;
108    painter.line_segment(
109        [Pos2::new(rect.min.x, mid_y), Pos2::new(rect.max.x, mid_y)],
110        Stroke::new(1.0_f32, Color32::from_rgb(30, 38, 55)),
111    );
112
113    if samples.is_empty() {
114        return;
115    }
116
117    let w = rect.width();
118    let h = rect.height() * 0.45;
119    let step_x = w / (samples.len() as f32);
120
121    let mut prev_pt = Pos2::new(rect.min.x, mid_y - (samples[0] * h));
122    for (i, &s) in samples.iter().enumerate().skip(1) {
123        let x = rect.min.x + (i as f32 * step_x);
124        let y = (mid_y - (s * h)).clamp(rect.min.y + 2.0, rect.max.y - 2.0);
125        let curr_pt = Pos2::new(x, y);
126        painter.line_segment([prev_pt, curr_pt], Stroke::new(1.5_f32, stroke_color));
127        prev_pt = curr_pt;
128    }
129}

131fn render_pwm_stream(ui: &mut egui::Ui, pulses: &[f32], pulse_color: Color32) {
132    let (response, painter) =
133        ui.allocate_painter(Vec2::new(ui.available_width(), 85.0), egui::Sense::hover());
134    let rect = response.rect;
135    painter.rect_filled(rect, 4.0, Color32::from_rgb(12, 15, 22));
136    painter.rect_stroke(
137        rect,
138        4.0,
139        Stroke::new(1.0_f32, Color32::from_rgb(35, 42, 60)),
140        StrokeKind::Inside,
141    );
142
143    if pulses.is_empty() {
144        return;
145    }
146
147    let num_display = 256.min(pulses.len());
148    let step_x = rect.width() / (num_display as f32);
149    let top_y = rect.min.y + 12.0;
150    let bot_y = rect.max.y - 12.0;
151
152    let start_idx = pulses.len().saturating_sub(num_display);
153    let mut prev_pt = Pos2::new(
154        rect.min.x,
155        if pulses[start_idx] > 0.0 {
156            top_y
157        } else {
158            bot_y
159        },
160    );
161
162    for i in 0..num_display {
163        let val = pulses[start_idx + i];
164        let x = rect.min.x + (i as f32 * step_x);
165        let target_y = if val > 0.0 { top_y } else { bot_y };
166
167        let curr_pt = Pos2::new(x, target_y);
168        painter.line_segment(
169            [prev_pt, Pos2::new(x, prev_pt.y)],
170            Stroke::new(1.2_f32, pulse_color),
171        );
172        painter.line_segment(
173            [Pos2::new(x, prev_pt.y), curr_pt],
174            Stroke::new(1.2_f32, pulse_color),
175        );
176        prev_pt = curr_pt;
177    }
178}

180fn render_dma_buffer_monitor(ui: &mut egui::Ui, current_step: u32, hw: &HardwareTargetConfig) {
181    let is_half_0 = (current_step % 2) == 0;
182    ui.horizontal(|ui| {
183        // Buffer 0
184        let col_0 = if is_half_0 {
185            Color32::from_rgb(40, 160, 90)
186        } else {
187            Color32::from_rgb(30, 45, 60)
188        };
189        let text_0 = if is_half_0 {
190            "DMA Half 0: STREAMING ▶"
191        } else {
192            "DMA Half 0: READY"
193        };
194        ui.label(egui::RichText::new(text_0).color(col_0).monospace().small());
195
196        ui.separator();
197
198        // Buffer 1
199        let col_1 = if !is_half_0 {
200            Color32::from_rgb(220, 140, 30)
201        } else {
202            Color32::from_rgb(30, 45, 60)
203        };
204        let text_1 = if !is_half_0 {
205            "DMA Half 1: DSP FILLING ⚡"
206        } else {
207            "DMA Half 1: READY"
208        };
209        ui.label(egui::RichText::new(text_1).color(col_1).monospace().small());
210
211        ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {
212            ui.label(
213                egui::RichText::new(format!("{} Words / Transfer", hw.dma_buffer_size)).small(),
214            );
215        });
216    });
217}

219fn render_piezo_acoustic_curve(
220    ui: &mut egui::Ui,
221    eq: &embedded_audio_codegen::PiezoAcousticConfig,
222) {
223    let (response, painter) =
224        ui.allocate_painter(Vec2::new(ui.available_width(), 85.0), egui::Sense::hover());
225    let rect = response.rect;
226    painter.rect_filled(rect, 4.0, Color32::from_rgb(12, 15, 22));
227    painter.rect_stroke(
228        rect,
229        4.0,
230        Stroke::new(1.0_f32, Color32::from_rgb(35, 42, 60)),
231        StrokeKind::Inside,
232    );
233
234    // Center reference 0 dB line
235    let mid_y = rect.min.y + 40.0;
236    painter.line_segment(
237        [Pos2::new(rect.min.x, mid_y), Pos2::new(rect.max.x, mid_y)],
238        Stroke::new(1.0_f32, Color32::from_rgb(40, 48, 65)),
239    );
240
241    let num_points = 64;
242    let step_x = rect.width() / (num_points as f32);
243
244    let mut prev_raw = Pos2::new(rect.min.x, mid_y);
245    let mut prev_flat = Pos2::new(rect.min.x, mid_y);
246
247    for i in 0..num_points {
248        let frac = i as f32 / num_points as f32;
249        let freq = 100.0 * (100.0_f32).powf(frac); // 100Hz to 10kHz log scale
250
251        // Model raw piezo acoustic resonance peak at resonance_freq_hz
252        let delta_f = (freq - eq.resonance_freq_hz) / 800.0;
253        let raw_peak_db = 16.0 / (1.0 + delta_f * delta_f) - 6.0;
254
255        let comp_db = if eq.enabled {
256            let notch_db = eq.notch_depth_db / (1.0 + delta_f * delta_f);
257            raw_peak_db + notch_db + (if freq < 800.0 { eq.bass_boost_db } else { 0.0 })
258        } else {
259            raw_peak_db
260        };
261
262        let x = rect.min.x + (i as f32 * step_x);
263        let raw_y = (mid_y - raw_peak_db * 2.0).clamp(rect.min.y + 3.0, rect.max.y - 3.0);
264        let flat_y = (mid_y - comp_db * 2.0).clamp(rect.min.y + 3.0, rect.max.y - 3.0);
265
266        let pt_raw = Pos2::new(x, raw_y);
267        let pt_flat = Pos2::new(x, flat_y);
268
269        if i > 0 {
270            // Draw Raw Piezo Spiky Curve in Coral Red
271            painter.line_segment(
272                [prev_raw, pt_raw],
273                Stroke::new(1.0_f32, Color32::from_rgba_unmultiplied(255, 80, 80, 140)),
274            );
275            // Draw DSP Flattened Acoustic Curve in Neon Green
276            painter.line_segment(
277                [prev_flat, pt_flat],
278                Stroke::new(1.8_f32, Color32::from_rgb(80, 240, 140)),
279            );
280        }
281
282        prev_raw = pt_raw;
283        prev_flat = pt_flat;
284    }
285
286    painter.text(
287        Pos2::new(rect.min.x + 8.0, rect.min.y + 8.0),
288        egui::Align2::LEFT_TOP,
289        "Red: Raw Piezo Peak  |  Green: DSP Compensated",
290        egui::FontId::monospace(10.0),
291        Color32::from_rgb(180, 200, 220),
292    );
293}

295fn render_spectrum(ui: &mut egui::Ui, mags: &[f32]) {
296    let (response, painter) =
297        ui.allocate_painter(Vec2::new(ui.available_width(), 75.0), egui::Sense::hover());
298    let rect = response.rect;
299    painter.rect_filled(rect, 4.0, Color32::from_rgb(12, 15, 22));
300    painter.rect_stroke(
301        rect,
302        4.0,
303        Stroke::new(1.0_f32, Color32::from_rgb(35, 42, 60)),
304        StrokeKind::Inside,
305    );
306
307    if mags.is_empty() {
308        return;
309    }
310
311    let num_bars = 48.min(mags.len());
312    let bar_w = (rect.width() / num_bars as f32) - 1.5;
313
314    for i in 0..num_bars {
315        let mag = mags[i].clamp(0.0, 1.0);
316        let bar_h = mag * (rect.height() - 10.0);
317        let x = rect.min.x + (i as f32 * (bar_w + 1.5)) + 2.0;
318        let y = rect.max.y - 5.0 - bar_h;
319
320        let color = if mag > 0.8 {
321            Color32::from_rgb(255, 90, 90)
322        } else if mag > 0.4 {
323            Color32::from_rgb(255, 200, 60)
324        } else {
325            Color32::from_rgb(90, 210, 140)
326        };
327
328        painter.rect_filled(
329            Rect::from_min_size(Pos2::new(x, y), Vec2::new(bar_w, bar_h.max(2.0))),
330            1.5,
331            color,
332        );
333    }
334}

336fn render_mcu_profiler(ui: &mut egui::Ui, project: &DawProject) {
337    let mut total_notes = 0;
338    for t in &project.tracks {
339        total_notes += t.notes.len();
340    }
341
342    // Footprint estimations
343    let bank_header_bytes = 16;
344    let inst_bytes = project.instruments.len() * 24;
345    let note_bytes = total_notes * 4;
346    let total_rom_estimate = bank_header_bytes + inst_bytes + note_bytes;
347    let ram_estimate_bytes =
348        (project.tracks.len() * 32) + (project.hardware_target.dma_buffer_size * 2);
349
350    ui.horizontal(|ui| {
351        ui.label("Flash ROM Footprint:");
352        ui.label(
353            egui::RichText::new(format!(
354                "{} B ({:.2} KB)",
355                total_rom_estimate,
356                total_rom_estimate as f32 / 1024.0
357            ))
358            .color(Color32::GREEN),
359        );
360        ui.separator();
361        ui.label("RAM Usage:");
362        ui.label(
363            egui::RichText::new(format!("{} B", ram_estimate_bytes)).color(Color32::LIGHT_BLUE),
364        );
365    });
366
367    ui.add_space(2.0);
368
369    // Target MCU specific load
370    let (target_name, mcu_mhz, mcu_scale) = match project.hardware_target.target_mcu {
371        TargetMcu::Stm32U5 => ("STM32U5 (Cortex-M33)", 160, 0.22),
372        TargetMcu::Stm32F4 => ("STM32F4 (Cortex-M4)", 168, 0.28),
373        TargetMcu::Rp2040 => ("RP2040 (Cortex-M0+)", 125, 0.65),
374        TargetMcu::Esp32RiscV => ("ESP32-C3 (RISC-V)", 160, 0.35),
375        TargetMcu::GenericCortexM0 => ("Generic Cortex-M0+", 48, 1.8),
376    };
377
378    let est_load = (project.tracks.len() as f32 * mcu_scale).min(100.0);
379    ui.horizontal(|ui| {
380        ui.label(format!(
381            "Target CPU Load ({} @ {}MHz):",
382            target_name, mcu_mhz
383        ));
384        ui.label(
385            egui::RichText::new(format!("{:.1}%", est_load))
386                .color(Color32::from_rgb(100, 240, 150))
387                .strong(),
388        );
389    });
