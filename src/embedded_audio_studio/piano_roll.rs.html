1//! Modern High-Contrast Step-Sequencer and Piano-Roll timeline editor with full-range multi-octave visibility.

3use eframe::egui::{self, Color32, Pos2, Rect, Stroke, StrokeKind, Vec2};
4use embedded_audio_codegen::{NoteEvent, Track};
5
6/// Distinct high-contrast palette for tracks.
7pub const TRACK_COLORS: [Color32; 5] = [
8    Color32::from_rgb(60, 170, 255),  // Electric Blue (Lead)
9    Color32::from_rgb(60, 230, 140),  // Emerald Green (Bass)
10    Color32::from_rgb(255, 165, 45),  // Amber Gold (Percussion/SFX)
11    Color32::from_rgb(205, 100, 255), // Neon Purple (FM/Wavetable)
12    Color32::from_rgb(255, 85, 150),  // Hot Pink (Arp)
13];
14
15pub struct PianoRollState {
16    pub selected_track_idx: usize,
17    pub base_midi_note: u8,
18    pub num_notes_displayed: usize,
19    pub current_velocity: u8,
20    pub step_width: f32,
21    pub key_height: f32,
22    pub show_ghost_notes: bool,
23}
24
25impl Default for PianoRollState {
26    fn default() -> Self {
27        Self {
28            selected_track_idx: 0,
29            base_midi_note: 24,      // C1 (covers sub-bass and explosions)
30            num_notes_displayed: 72, // 6 full octaves: C1 to B6 (covers all bass, leads, and hi-hats!)
31            current_velocity: 100,
32            step_width: 32.0,
33            key_height: 18.0,
34            show_ghost_notes: true,
35        }
36    }
37}
38
39impl PianoRollState {
40    /// Center the view on the pitch range of the given track.
41    pub fn auto_fit_octave(&mut self, track: &Track) {
42        if track.notes.is_empty() {
43            return;
44        }
45        let min_n = track.notes.iter().map(|n| n.note).min().unwrap_or(24);
46        let max_n = track.notes.iter().map(|n| n.note).max().unwrap_or(95);
47
48        // Set base note so all notes are visible
49        let target_base = (min_n / 12 * 12).saturating_sub(12).clamp(12, 60);
50        self.base_midi_note = target_base;
51        let needed_range = (max_n as usize + 12)
52            .saturating_sub(target_base as usize)
53            .max(48);
54        self.num_notes_displayed = needed_range.min(84);
55    }
56
57    /// Fit all notes across all tracks in the project.
58    pub fn auto_fit_all_tracks(&mut self, tracks: &[Track]) {
59        let mut min_n = 127;
60        let mut max_n = 0;
61        let mut has_notes = false;
62
63        for t in tracks {
64            for n in &t.notes {
65                has_notes = true;
66                if n.note < min_n {
67                    min_n = n.note;
68                }
69                if n.note > max_n {
70                    max_n = n.note;
71                }
72            }
73        }
74
75        if has_notes {
76            self.base_midi_note = (min_n / 12 * 12).clamp(12, 60);
77            let needed = (max_n as usize + 12)
78                .saturating_sub(self.base_midi_note as usize)
79                .max(48);
80            self.num_notes_displayed = needed.min(84);
81        } else {
82            self.base_midi_note = 24;
83            self.num_notes_displayed = 72;
84        }
85    }
86}

88pub fn render_piano_roll(
89    ui: &mut egui::Ui,
90    state: &mut PianoRollState,
91    tracks: &mut [Track],
92    total_steps: u32,
93    steps_per_beat: u32,
94    current_playing_step: u32,
95) {
96    ui.vertical(|ui| {
97        // Track selector tabs & track settings
98        let mut solo_clicked_idx = None;
99        ui.horizontal_wrapped(|ui| {
100            ui.label(egui::RichText::new("Tracks:").strong());
101            for (idx, track) in tracks.iter_mut().enumerate() {
102                let is_selected = idx == state.selected_track_idx;
103                let track_color = TRACK_COLORS[idx % TRACK_COLORS.len()];
104
105                let btn_text = egui::RichText::new(format!("● {}", track.name))
106                    .color(if is_selected {
107                        Color32::WHITE
108                    } else if track.muted {
109                        Color32::GRAY
110                    } else {
111                        track_color
112                    })
113                    .strong();
114
115                let btn = egui::Button::new(btn_text).fill(if is_selected {
116                    track_color.gamma_multiply(0.6)
117                } else {
118                    Color32::from_rgb(25, 30, 42)
119                });
120
121                if ui.add(btn).clicked() {
122                    state.selected_track_idx = idx;
123                }
124
125                // Mute button (M)
126                let m_text = egui::RichText::new(if track.muted { "MUTE" } else { "M" })
127                    .color(if track.muted {
128                        Color32::WHITE
129                    } else {
130                        Color32::from_rgb(160, 170, 185)
131                    })
132                    .small()
133                    .strong();
134                let m_btn = egui::Button::new(m_text).fill(if track.muted {
135                    Color32::from_rgb(220, 50, 50)
136                } else {
137                    Color32::from_rgb(32, 38, 50)
138                });
139                if ui.add(m_btn).clicked() {
140                    track.muted = !track.muted;
141                }
142
143                // Solo button (S) - Standard exclusive solo (or Shift-click for multi-solo)
144                let s_text = egui::RichText::new(if track.solo { "SOLO" } else { "S" })
145                    .color(if track.solo {
146                        Color32::BLACK
147                    } else {
148                        Color32::from_rgb(160, 170, 185)
149                    })
150                    .small()
151                    .strong();
152                let s_btn = egui::Button::new(s_text).fill(if track.solo {
153                    Color32::from_rgb(240, 200, 40)
154                } else {
155                    Color32::from_rgb(32, 38, 50)
156                });
157                if ui.add(s_btn).clicked() {
158                    solo_clicked_idx = Some(idx);
159                }
160
161                ui.add_space(8.0);
162            }
163        });
164
165        // Apply exclusive solo logic
166        if let Some(s_idx) = solo_clicked_idx {
167            let was_solo = tracks.get(s_idx).map(|t| t.solo).unwrap_or(false);
168            let shift_held = ui.input(|i| i.modifiers.shift || i.modifiers.ctrl);
169            if was_solo {
170                if let Some(t) = tracks.get_mut(s_idx) {
171                    t.solo = false;
172                }
173            } else if shift_held {
174                if let Some(t) = tracks.get_mut(s_idx) {
175                    t.solo = true;
176                }
177            } else {
178                for (i, t) in tracks.iter_mut().enumerate() {
179                    t.solo = i == s_idx;
180                }
181            }
182        }
183
184        ui.add_space(4.0);
185
186        // Track Details Toolbar & Range Zoom Controls
187        let active_track_color = TRACK_COLORS[state.selected_track_idx % TRACK_COLORS.len()];
188        let mut do_fit_all = false;
189        let mut toolbar_solo_toggled = false;
190        if let Some(track) = tracks.get_mut(state.selected_track_idx) {
191            ui.horizontal_wrapped(|ui| {
192                ui.label(
193                    egui::RichText::new(format!("Track: {}", track.name))
194                        .color(active_track_color)
195                        .strong(),
196                );
197                ui.add_space(6.0);
198
199                // Quick Mute & Solo toggle for active track
200                let is_muted = track.muted;
201                ui.checkbox(
202                    &mut track.muted,
203                    egui::RichText::new("Mute").color(if is_muted {
204                        Color32::from_rgb(255, 90, 90)
205                    } else {
206                        Color32::WHITE
207                    }),
208                );
209                let is_solo = track.solo;
210                let mut new_solo = is_solo;
211                if ui
212                    .checkbox(
213                        &mut new_solo,
214                        egui::RichText::new("Solo").color(if is_solo {
215                            Color32::from_rgb(255, 220, 60)
216                        } else {
217                            Color32::WHITE
218                        }),
219                    )
220                    .changed()
221                {
222                    toolbar_solo_toggled = true;
223                }
224
225                ui.add_space(6.0);
226                ui.separator();
227                ui.add_space(6.0);
228
229                ui.label("Vol:");
230                let mut vol = track.volume_q8 as f32 / 2.55;
231                if ui
232                    .add(egui::Slider::new(&mut vol, 0.0..=100.0).suffix("%"))
233                    .changed()
234                {
235                    track.volume_q8 = (vol * 2.55) as u8;
236                }
237
238                ui.add_space(6.0);
239                ui.label("Vel:");
240                ui.add(egui::Slider::new(&mut state.current_velocity, 1..=127));
241
242                ui.add_space(6.0);
243                ui.separator();
244                ui.add_space(6.0);
245
246                // Octave & Range View Presets
247                ui.label(egui::RichText::new("Range:").strong());
248                let min_note_name = format!("C{}", state.base_midi_note / 12 - 1);
249                let max_note_name = format!(
250                    "B{}",
251                    (state.base_midi_note as usize + state.num_notes_displayed) / 12 - 1
252                );
253                ui.label(
254                    egui::RichText::new(format!("{} - {}", min_note_name, max_note_name))
255                        .strong()
256                        .color(Color32::from_rgb(255, 230, 100)),
257                );
258
259                let mut do_fit_track = false;
260
261                if ui.button("🔍 Fit All").clicked() {
262                    do_fit_all = true;
263                }
264
265                if ui.button("🎯 Fit Track").clicked() {
266                    do_fit_track = true;
267                }
268
269                if ui.button("▲ Shift Up").clicked() && state.base_midi_note < 60 {
270                    state.base_midi_note += 12;
271                }
272                if ui.button("▼ Shift Down").clicked() && state.base_midi_note >= 12 {
273                    state.base_midi_note -= 12;
274                }
275
276                // Vertical Zoom controls
277                if ui.button("➕ Zoom Y").clicked() && state.key_height < 32.0 {
278                    state.key_height += 2.0;
279                }
280                if ui.button("➖ Zoom Y").clicked() && state.key_height > 12.0 {
281                    state.key_height -= 2.0;
282                }
283
284                ui.checkbox(&mut state.show_ghost_notes, "Ghost Tracks");
285
286                ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {
287                    if ui.button("🗑 Clear Track").clicked() {
288                        track.notes.clear();
289                    }
290                    ui.label(format!("Notes: {}", track.notes.len()));
291                });
292
293                if do_fit_track {
294                    state.auto_fit_octave(track);
295                }
296            });
297        }
298
299        if do_fit_all {
300            state.auto_fit_all_tracks(tracks);
301        }
302
303        if toolbar_solo_toggled {
304            let s_idx = state.selected_track_idx;
305            let was_solo = tracks.get(s_idx).map(|t| t.solo).unwrap_or(false);
306            if was_solo {
307                if let Some(t) = tracks.get_mut(s_idx) {
308                    t.solo = false;
309                }
310            } else {
311                for (i, t) in tracks.iter_mut().enumerate() {
312                    t.solo = i == s_idx;
313                }
314            }
315        }
316
317        ui.add_space(4.0);
318
319        // Check if any notes in the selected track fall outside visible bounds
320        let base_midi_note = state.base_midi_note;
321        let num_notes_displayed = state.num_notes_displayed;
322        let top_midi_note = base_midi_note as usize + num_notes_displayed;
323
324        if let Some(track) = tracks.get(state.selected_track_idx) {
325            let notes_above = track
326                .notes
327                .iter()
328                .filter(|n| n.note as usize >= top_midi_note)
329                .count();
330            let notes_below = track
331                .notes
332                .iter()
333                .filter(|n| (n.note as usize) < base_midi_note as usize)
334                .count();
335
336            if notes_above > 0 || notes_below > 0 {
337                ui.horizontal(|ui| {
338                    if notes_above > 0 {
339                        if ui
340                            .button(
341                                egui::RichText::new(format!(
342                                    "▲ {} notes above view (Click to expand view)",
343                                    notes_above
344                                ))
345                                .color(Color32::from_rgb(255, 200, 60)),
346                            )
347                            .clicked()
348                        {
349                            state.num_notes_displayed = (state.num_notes_displayed + 24).min(84);
350                        }
351                    }
352                    if notes_below > 0 {
353                        if ui
354                            .button(
355                                egui::RichText::new(format!(
356                                    "▼ {} notes below view (Click to expand view)",
357                                    notes_below
358                                ))
359                                .color(Color32::from_rgb(255, 200, 60)),
360                            )
361                            .clicked()
362                        {
363                            state.base_midi_note = state.base_midi_note.saturating_sub(12);
364                            state.num_notes_displayed = (state.num_notes_displayed + 12).min(84);
365                        }
366                    }
367                });
368                ui.add_space(2.0);
369            }
370        }
371
372        // Find which MIDI notes are currently actively hitting at `current_playing_step`
373        let mut hitting_notes: Vec<(usize, u8)> = Vec::new(); // (track_idx, note)
374        for (t_idx, t) in tracks.iter().enumerate() {
375            if !t.muted {
376                for n in &t.notes {
377                    if current_playing_step >= n.step
378                        && current_playing_step < n.step + n.duration_steps
379                    {
380                        hitting_notes.push((t_idx, n.note));
381                    }
382                }
383            }
384        }
385
386        // Scroll Area containing Full Multi-Octave Timeline Canvas
387        egui::ScrollArea::both()
388            .auto_shrink([false, false])
389            .max_height(ui.available_height() - 10.0)
390            .show(ui, |ui| {
391                let total_width = 85.0 + (total_steps as f32 * state.step_width);
392                let total_height = 26.0 + (num_notes_displayed as f32 * state.key_height);
393
394                let (response, painter) = ui.allocate_painter(
395                    Vec2::new(total_width, total_height),
396                    egui::Sense::click_and_drag(),
397                );
398                let origin = response.rect.min;
399
400                // 1. Draw Background Timeline Header (Steps and Beats)
401                let header_rect = Rect::from_min_size(origin, Vec2::new(total_width, 26.0));
402                painter.rect_filled(header_rect, 0.0, Color32::from_rgb(18, 22, 32));
403
404                for step in 0..total_steps {
405                    let x = origin.x + 85.0 + (step as f32 * state.step_width);
406                    let is_beat = (step % steps_per_beat) == 0;
407
408                    let line_color = if is_beat {
409                        Color32::from_rgb(70, 88, 125)
410                    } else {
411                        Color32::from_rgb(32, 40, 58)
412                    };
413
414                    painter.line_segment(
415                        [
416                            Pos2::new(x, origin.y),
417                            Pos2::new(x, origin.y + total_height),
418                        ],
419                        Stroke::new(if is_beat { 1.5_f32 } else { 1.0_f32 }, line_color),
420                    );
421
422                    if is_beat {
423                        painter.text(
424                            Pos2::new(x + 4.0, origin.y + 13.0),
425                            egui::Align2::LEFT_CENTER,
426                            format!("{}", (step / steps_per_beat) + 1),
427                            egui::FontId::monospace(12.0),
428                            Color32::from_rgb(200, 220, 255),
429                        );
430                    }
431                }
432
433                // 2. Draw Active Step Column Highlight Beam (Playhead background column)
434                if current_playing_step < total_steps {
435                    let beam_x = origin.x + 85.0 + (current_playing_step as f32 * state.step_width);
436                    let beam_rect = Rect::from_min_size(
437                        Pos2::new(beam_x, origin.y),
438                        Vec2::new(state.step_width, total_height),
439                    );
440                    painter.rect_filled(
441                        beam_rect,
442                        0.0,
443                        Color32::from_rgba_unmultiplied(255, 220, 80, 28),
444                    );
445                }
446
447                // 3. Draw Piano Keys on Left and Grid Rows
448                let note_names = [
449                    "C", "C#", "D", "D#", "E", "F", "F#", "G", "G#", "A", "A#", "B",
450                ];
451                for row in 0..num_notes_displayed {
452                    let midi_note = base_midi_note + (num_notes_displayed - 1 - row) as u8;
453                    let y = origin.y + 26.0 + (row as f32 * state.key_height);
454                    let note_idx = (midi_note % 12) as usize;
455                    let is_black = [1, 3, 6, 8, 10].contains(&note_idx);
456
457                    // Row background with high contrast
458                    let row_rect = Rect::from_min_size(
459                        Pos2::new(origin.x + 85.0, y),
460                        Vec2::new(total_width - 85.0, state.key_height),
461                    );
462                    let row_bg = if is_black {
463                        Color32::from_rgb(14, 17, 25)
464                    } else {
465                        Color32::from_rgb(22, 27, 40)
466                    };
467                    painter.rect_filled(row_rect, 0.0, row_bg);
468
469                    // Octave boundary line (C note top border)
470                    let is_c_boundary = note_idx == 0;
471                    painter.line_segment(
472                        [
473                            Pos2::new(origin.x, y + state.key_height),
474                            Pos2::new(origin.x + total_width, y + state.key_height),
475                        ],
476                        Stroke::new(
477                            if is_c_boundary { 1.8_f32 } else { 1.0_f32 },
478                            if is_c_boundary {
479                                Color32::from_rgb(60, 80, 120)
480                            } else {
481                                Color32::from_rgb(34, 42, 60)
482                            },
483                        ),
484                    );
485
486                    // Check if this key is currently hitting right now!
487                    let is_hitting = hitting_notes.iter().any(|(_, n)| *n == midi_note);
488
489                    // Piano Key button on left
490                    let key_rect = Rect::from_min_size(
491                        Pos2::new(origin.x, y),
492                        Vec2::new(85.0, state.key_height),
493                    );
494                    let key_color = if is_hitting {
495                        Color32::from_rgb(255, 220, 60) // Bright Gold Active Glow
496                    } else if is_black {
497                        Color32::from_rgb(32, 38, 50)
498                    } else {
499                        Color32::from_rgb(220, 230, 245)
500                    };
501
502                    let key_text_color = if is_hitting {
503                        Color32::BLACK
504                    } else if is_black {
505                        Color32::WHITE
506                    } else {
507                        Color32::BLACK
508                    };
509
510                    painter.rect_filled(key_rect, 2.0, key_color);
511                    let label = format!("{}{}", note_names[note_idx], (midi_note / 12) as i32 - 1);
512                    painter.text(
513                        Pos2::new(origin.x + 8.0, y + state.key_height * 0.5),
514                        egui::Align2::LEFT_CENTER,
515                        label,
516                        egui::FontId::proportional(11.0),
517                        key_text_color,
518                    );
519                }
520
521                // 4. Render Ghost Notes (Other Tracks) for Full Arrangement View
522                if state.show_ghost_notes {
523                    for (t_idx, other_track) in tracks.iter().enumerate() {
524                        if t_idx == state.selected_track_idx || other_track.muted {
525                            continue;
526                        }
527                        let ghost_color =
528                            TRACK_COLORS[t_idx % TRACK_COLORS.len()].gamma_multiply(0.35);
529
530                        for note in &other_track.notes {
531                            if note.note >= base_midi_note
532                                && (note.note as usize)
533                                    < (base_midi_note as usize + num_notes_displayed)
534                            {
535                                let row = (num_notes_displayed - 1)
536                                    - (note.note - base_midi_note) as usize;
537                                let nx = origin.x + 85.0 + (note.step as f32 * state.step_width);
538                                let ny = origin.y + 26.0 + (row as f32 * state.key_height);
539                                let nw =
540                                    (note.duration_steps as f32 * state.step_width).max(12.0) - 2.0;
541                                let nh = state.key_height - 2.0;
542
543                                let note_rect = Rect::from_min_size(
544                                    Pos2::new(nx + 1.0, ny + 1.0),
545                                    Vec2::new(nw, nh),
546                                );
547                                painter.rect_filled(note_rect, 2.0, ghost_color);
548                            }
549                        }
550                    }
551                }
552
553                // 5. Render Notes for Selected Active Track
554                if let Some(track) = tracks.get_mut(state.selected_track_idx) {
555                    let mut note_to_remove = None;
556
557                    for (idx, note) in track.notes.iter().enumerate() {
558                        if note.note >= base_midi_note
559                            && (note.note as usize)
560                                < (base_midi_note as usize + num_notes_displayed)
561                        {
562                            let row =
563                                (num_notes_displayed - 1) - (note.note - base_midi_note) as usize;
564                            let nx = origin.x + 85.0 + (note.step as f32 * state.step_width);
565                            let ny = origin.y + 26.0 + (row as f32 * state.key_height);
566                            let nw =
567                                (note.duration_steps as f32 * state.step_width).max(12.0) - 2.0;
568                            let nh = state.key_height - 2.0;
569
570                            let note_rect = Rect::from_min_size(
571                                Pos2::new(nx + 1.0, ny + 1.0),
572                                Vec2::new(nw, nh),
573                            );
574
575                            // Check if this specific note is currently being played by playhead
576                            let is_active_hit = current_playing_step >= note.step
577                                && current_playing_step < note.step + note.duration_steps;
578
579                            let (fill_col, stroke_col, stroke_w) = if is_active_hit {
580                                (
581                                    Color32::from_rgb(255, 235, 90),  // Blazing Gold on Note Hit!
582                                    Color32::from_rgb(255, 255, 255), // White Glow Border
583                                    2.5_f32,
584                                )
585                            } else {
586                                (
587                                    active_track_color,
588                                    Color32::from_rgb(220, 240, 255),
589                                    1.0_f32,
590                                )
591                            };
592
593                            painter.rect_filled(note_rect, 3.0, fill_col);
594                            painter.rect_stroke(
595                                note_rect,
596                                3.0,
597                                Stroke::new(stroke_w, stroke_col),
598                                StrokeKind::Inside,
599                            );
600
601                            // Note pitch name label inside the note box
602                            let note_name = note_names[(note.note % 12) as usize];
603                            painter.text(
604                                Pos2::new(nx + 4.0, ny + state.key_height * 0.5),
605                                egui::Align2::LEFT_CENTER,
606                                format!("{}{}", note_name, (note.note / 12) as i32 - 1),
607                                egui::FontId::monospace(10.0),
608                                Color32::BLACK,
609                            );
610
611                            // Right click on note to delete
612                            if response.clicked_by(egui::PointerButton::Secondary) {
613                                if let Some(mouse_pos) = response.hover_pos() {
614                                    if note_rect.contains(mouse_pos) {
615                                        note_to_remove = Some(idx);
616                                    }
617                                }
618                            }
619                        }
620                    }
621
622                    if let Some(rem_idx) = note_to_remove {
623                        track.notes.remove(rem_idx);
624                    }
625
626                    // Left Click to place or remove a note
627                    if response.clicked_by(egui::PointerButton::Primary) {
628                        if let Some(mouse_pos) = response.hover_pos() {
629                            if mouse_pos.x > origin.x + 85.0 && mouse_pos.y > origin.y + 26.0 {
630                                let step =
631                                    ((mouse_pos.x - (origin.x + 85.0)) / state.step_width) as u32;
632                                let row =
633                                    ((mouse_pos.y - (origin.y + 26.0)) / state.key_height) as usize;
634                                if step < total_steps && row < num_notes_displayed {
635                                    let midi_note =
636                                        base_midi_note + (num_notes_displayed - 1 - row) as u8;
637
638                                    // Remove existing note at same step & pitch if any, otherwise add
639                                    if let Some(pos) = track
640                                        .notes
641                                        .iter()
642                                        .position(|n| n.step == step && n.note == midi_note)
643                                    {
644                                        track.notes.remove(pos);
645                                    } else {
646                                        track.notes.push(NoteEvent {
647                                            step,
648                                            note: midi_note,
649                                            duration_steps: 2,
650                                            velocity: state.current_velocity,
651                                        });
652                                    }
653                                }
654                            }
655                        }
656                    }
657                }
658
659                // 6. Draw High-Visibility Playhead Line & Top Marker
660                if current_playing_step < total_steps {
661                    let playhead_x =
662                        origin.x + 85.0 + (current_playing_step as f32 * state.step_width);
663
664                    // Solid bright red needle
665                    painter.line_segment(
666                        [
667                            Pos2::new(playhead_x, origin.y),
668                            Pos2::new(playhead_x, origin.y + total_height),
669                        ],
670                        Stroke::new(3.0_f32, Color32::from_rgb(255, 60, 60)),
671                    );
672
673                    // Header marker needle
674                    painter.rect_filled(
675                        Rect::from_min_size(
676                            Pos2::new(playhead_x - 3.0, origin.y),
677                            Vec2::new(6.0, 10.0),
678                        ),
679                        2.0,
680                        Color32::from_rgb(255, 90, 90),
681                    );
682                }
683            });
684    });
