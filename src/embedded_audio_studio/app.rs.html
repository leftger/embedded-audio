15pub enum ActiveTab {
16    PianoRoll,
17    SynthLab,
18    HardwareVisualizers,
19    DeviceLink,
20    Exporter,
21}

23pub struct EmbeddedAudioStudioApp {

36impl EmbeddedAudioStudioApp {
37    pub fn new(_cc: &eframe::CreationContext<'_>) -> Self {
38        let project = DawProject::chiptune_odyssey();
39        let (audio_device, visualizer_data) = HostAudioDevice::new(project.clone());
40
41        let audio_state = Arc::clone(&audio_device.state);
42        let mut piano_roll_state = PianoRollState::default();
43        piano_roll_state.auto_fit_all_tracks(&project.tracks);
44
45        Self {
46            active_tab: ActiveTab::PianoRoll,
47            project,
48            _audio_device: audio_device,
49            audio_state,
50            visualizer_data,
51            piano_roll_state,
52            synth_lab_state: SynthLabState::default(),
53            device_link_state: DeviceLinkState::default(),
54            exporter_state: ExporterState::default(),
55            selected_preset_idx: 0,
56        }
57    }
58
59    pub fn load_preset(&mut self, idx: usize) {
60        self.selected_preset_idx = idx;
61        let new_project = match idx {
62            0 => DawProject::chiptune_odyssey(),
63            1 => DawProject::fm_cyberpunk(),
64            2 => DawProject::boss_battle(),
65            3 => DawProject::lofi_nostalgia(),
66            4 => DawProject::sfx_showcase(),
67            _ => DawProject::chiptune_odyssey(),
68        };
69
70        self.project = new_project.clone();
71        self.piano_roll_state.selected_track_idx = 0;
72        self.synth_lab_state.selected_inst_idx = 0;
73        self.piano_roll_state
74            .auto_fit_all_tracks(&self.project.tracks);
75
76        if let Ok(mut host) = self.audio_state.lock() {
77            host.stop();
78            host.set_project(new_project);
79        }
80    }
81}
