1//! Hardware-in-the-loop device connection and live audio streaming bridge.
2
3use eframe::egui::{self, Color32};
4use embedded_audio_live::MockDeviceBridge;
5
6pub struct DeviceLinkState {
7    pub is_connected: bool,
8    pub transport_type: usize, // 0 = Loopback Mock, 1 = USB CDC, 2 = UART Serial
9    pub port_name: String,
10    pub baud_rate: u32,
11    pub mock_bridge: MockDeviceBridge,
12    pub last_telemetry: Option<(u8, u32, u32)>, // (cpu_pct, underruns, free_heap)
13    pub log_messages: Vec<String>,
14}
15
16impl Default for DeviceLinkState {
17    fn default() -> Self {
18        Self {
19            is_connected: false,
20            transport_type: 0,
21            port_name: "/dev/ttyACM0".to_string(),
22            baud_rate: 115200,
23            mock_bridge: MockDeviceBridge::new("RP2040 Embedded PWM Board"),
24            last_telemetry: Some((12, 0, 18432)),
25            log_messages: vec!["Ready to connect to embedded target.".to_string()],
26        }
27    }
28}
29
30pub fn render_device_link(ui: &mut egui::Ui, state: &mut DeviceLinkState) {
31    ui.vertical(|ui| {
32        ui.heading("Device Live-Link & Real-Time Hardware Auditioning");
33        ui.add_space(4.0);
34
35        ui.columns(2, |cols| {
36            // Left: Connection Settings
37            cols[0].group(|ui| {
38                ui.label(egui::RichText::new("Target Hardware Interface").strong());
39                ui.add_space(4.0);
40
41                ui.horizontal(|ui| {
42                    ui.label("Interface:");
43                    ui.selectable_value(&mut state.transport_type, 0, "Mock Loopback");
44                    ui.selectable_value(&mut state.transport_type, 1, "USB CDC");
45                    ui.selectable_value(&mut state.transport_type, 2, "UART Serial");
46                });
47
48                if state.transport_type != 0 {
49                    ui.horizontal(|ui| {
50                        ui.label("Port:");
51                        ui.text_edit_singleline(&mut state.port_name);
52                    });
53                    ui.horizontal(|ui| {
54                        ui.label("Baud Rate:");
55                        ui.selectable_value(&mut state.baud_rate, 115200, "115200");
56                        ui.selectable_value(&mut state.baud_rate, 921600, "921600 (High-Speed)");
57                    });
58                }
59
60                ui.add_space(6.0);
61
62                ui.horizontal(|ui| {
63                    if !state.is_connected {
64                        if ui.button(egui::RichText::new("⚡ Connect to Target").color(Color32::GREEN)).clicked() {
65                            state.is_connected = true;
66                            state.log_messages.push(format!("Connected to target via {}!", if state.transport_type == 0 { "Mock Transport" } else { &state.port_name }));
67                        }
68                    } else {
69                        if ui.button(egui::RichText::new("⏹ Disconnect").color(Color32::LIGHT_RED)).clicked() {
70                            state.is_connected = false;
71                            state.log_messages.push("Disconnected from target.".to_string());
72                        }
73                    }
74                });
75            });
76
77            // Right: Target Telemetry & Logs
78            cols[1].group(|ui| {
79                ui.label(egui::RichText::new("Device Telemetry & Health").strong());
80                ui.add_space(4.0);
81
82                if state.is_connected {
83                    ui.horizontal(|ui| {
84                        ui.label("Status:");
85                        ui.label(egui::RichText::new("ONLINE (STREAMING)").color(Color32::GREEN).strong());
86                    });
87                    ui.horizontal(|ui| {
88                        ui.label("Device:");
89                        ui.label(&state.mock_bridge.board_name);
90                    });
91                    if let Some((cpu, underruns, heap)) = state.last_telemetry {
92                        ui.horizontal(|ui| {
93                            ui.label("MCU CPU Load:");
94                            ui.label(egui::RichText::new(format!("{}%", cpu)).color(Color32::YELLOW));
95                        });
96                        ui.horizontal(|ui| {
97                            ui.label("Buffer Underruns:");
98                            ui.label(format!("{}", underruns));
99                        });
100                        ui.horizontal(|ui| {
101                            ui.label("Free RAM:");
102                            ui.label(format!("{} bytes", heap));
103                        });
104                    }
105                } else {
106                    ui.label(egui::RichText::new("Target offline. Connect to stream notes and audio directly to MCU pins.").italics());
107                }
108
109                ui.add_space(6.0);
110                ui.separator();
111                ui.label(egui::RichText::new("Activity Log:").small());
112                egui::ScrollArea::vertical().max_height(80.0).show(ui, |ui| {
113                    for msg in state.log_messages.iter().rev().take(6) {
114                        ui.label(egui::RichText::new(msg).monospace().small());
115                    }
116                });
117            });
118        });
119    });
