6/// Generates a standalone no_std Rust module with Embassy async tasks, timer drivers, and SFX channels.
7pub fn generate_embassy_code(project: &DawProject) -> String {
8    let mut code = String::new();
9    code.push_str("//! Auto-generated Embassy async audio driver by embedded-audio DAW studio.\n");
10    code.push_str(&format!(
11        "//! Target MCU: {:?} | Output Mode: {:?}\n",
12        project.hardware_target.target_mcu, project.hardware_target.output_mode
13    ));
14    code.push_str("//! Framework: embassy-rs (embassy-executor, embassy-time, embassy-sync)\n\n");
15    code.push_str("#![no_std]\n\n");
16    code.push_str("use embedded_audio::prelude::*;\n");
17    code.push_str("use embedded_audio::dsp::BiquadFilter;\n");
18    code.push_str("use embassy_executor::task;\n");
19    code.push_str("use embassy_time::{Duration, Timer};\n");
20    code.push_str("use embassy_sync::blocking_mutex::raw::CriticalSectionRawMutex;\n");
21    code.push_str("use embassy_sync::channel::Channel;\n\n");
22
23    // MCU-specific HAL imports
24    match project.hardware_target.target_mcu {
25        TargetMcu::Stm32U5 | TargetMcu::Stm32F4 => {
26            code.push_str("use embassy_stm32::timer::simple_pwm::SimplePwm;\n");
27            code.push_str("use embassy_stm32::time::khz;\n\n");
28        }
29        TargetMcu::Rp2040 => {
30            code.push_str("use embassy_rp::pwm::{Config as PwmConfig, Pwm};\n\n");
31        }
32        _ => {
33            code.push_str("// Generic embedded-hal / embassy timer bindings\n\n");
34        }
35    }
36
37    code.push_str(&format!(
38        "pub const SONG_TITLE: &str = \"{}\";\n",
39        project.title
40    ));
41    code.push_str(&format!("pub const SONG_BPM: u16 = {};\n", project.bpm));
42    code.push_str(&format!(
43        "pub const SAMPLE_RATE_HZ: u32 = {};\n",
44        project.sample_rate_hz
45    ));
46    code.push_str(&format!(
47        "pub const TOTAL_STEPS: u32 = {};\n",
48        project.total_steps
49    ));
50    code.push_str(&format!(
51        "pub const STEPS_PER_BEAT: u32 = {};\n\n",
52        project.steps_per_beat
53    ));
54
55    let samples_per_step =
56        (project.sample_rate_hz as u64 * 60) / (project.bpm as u64 * project.steps_per_beat as u64);
57    let step_micros = (60_000_000) / (project.bpm as u64 * project.steps_per_beat as u64);
58    code.push_str(&format!(
59        "pub const SAMPLES_PER_STEP: u32 = {};\n",
60        samples_per_step
61    ));
62    code.push_str(&format!(
63        "pub const STEP_DURATION_MICROS: u64 = {};\n\n",
64        step_micros
65    ));
66
67    // Note events structure & tables
68    code.push_str("/// Compact note event (step, note, duration, velocity)\n");
69    code.push_str("#[derive(Clone, Copy)]\n");
70    code.push_str("pub struct NoteEvent {\n");
71    code.push_str("    pub step: u16,\n");
72    code.push_str("    pub note: u8,\n");
73    code.push_str("    pub duration_steps: u8,\n");
74    code.push_str("    pub velocity: u8,\n");
75    code.push_str("}\n\n");
76
77    for track in &project.tracks {
78        let safe_name = track.name.replace([' ', '-', '(', ')'], "_").to_lowercase();
79        code.push_str(&format!(
80            "pub const TRACK_{}_NOTES: &[NoteEvent] = &[\n",
81            safe_name.to_uppercase()
82        ));
83        for n in &track.notes {
84            code.push_str(&format!(
85                "    NoteEvent {{ step: {}, note: {}, duration_steps: {}, velocity: {} }},\n",
86                n.step, n.note, n.duration_steps, n.velocity
87            ));
88        }
89        code.push_str("];\n\n");
90    }
91
92    // Piezo Acoustic Notch Filter parameters
93    let eq = &project.hardware_target.piezo_eq;
94    code.push_str("/// Piezo Acoustic Resonance Compensation EQ Parameters\n");
95    code.push_str(&format!(
96        "pub const PIEZO_EQ_ENABLED: bool = {};\n",
97        eq.enabled
98    ));
99    code.push_str(&format!(
100        "pub const PIEZO_RESONANCE_FREQ_HZ: f32 = {:.1};\n",
101        eq.resonance_freq_hz
102    ));
103    code.push_str(&format!(
104        "pub const PIEZO_NOTCH_DEPTH_DB: f32 = {:.1};\n",
105        eq.notch_depth_db
106    ));
107    code.push_str(&format!(
108        "pub const PWM_CARRIER_FREQ_KHZ: u32 = {};\n\n",
109        eq.carrier_freq_khz
110    ));
111
112    // Sound effect cue definitions and Async Channel
113    code.push_str(
114        r#"/// Async Sound Effect Event identifiers for Embassy channel dispatch.
115#[derive(Clone, Copy, Debug, PartialEq, Eq)]
116pub enum SfxCue {
117    Bootup,
118    LaserPew,
119    Shutdown,
120    CoinCollect,
121    PowerUpFanfare,
122    Explosion,
123    ErrorAlert,
124    Jump,
125}
126
127/// Global Async SFX Channel for non-blocking sound cue dispatch from any Embassy task.
128pub static SFX_CHANNEL: Channel<CriticalSectionRawMutex, SfxCue, 8> = Channel::new();
129
130/// Convenience async function to trigger a sound effect from buttons, sensors, or network tasks.
131pub async fn send_sfx(cue: SfxCue) {
132    SFX_CHANNEL.send(cue).await;
133}
134
135/// Try sending a sound effect without waiting (non-blocking).
136pub fn try_send_sfx(cue: SfxCue) -> Result<(), SfxCue> {
137    SFX_CHANNEL.try_send(cue).map_err(|e| match e {
138        embassy_sync::channel::TrySendError::Full(c) => c,
139    })
140}
141"#,
142    );
143
144    // Embassy Async Audio Task & Driver
145    code.push_str(
146        r#"
147/// Embassy Async Audio Playback Engine.
148pub struct EmbassyAudioEngine {
149    pub sigma_delta: SigmaDelta,
150    pub piezo_notch: BiquadFilter,
151    pub current_step: u16,
152    pub is_playing: bool,
153}
154
155impl EmbassyAudioEngine {
156    pub fn new() -> Self {
157        Self {
158            sigma_delta: SigmaDelta::new(),
159            piezo_notch: BiquadFilter::notch(PIEZO_RESONANCE_FREQ_HZ, 4.0, SAMPLE_RATE_HZ as f32),
160            current_step: 0,
161            is_playing: false,
162        }
163    }
164
165    pub fn start_song(&mut self) {
166        self.current_step = 0;
167        self.is_playing = true;
168    }
169
170    pub fn stop_song(&mut self) {
171        self.is_playing = false;
172        self.current_step = 0;
173    }
174
175    /// Process a single audio frame with optional Piezo notch filtering and 1-bit PDM output.
176    pub fn process_sample(&mut self, raw_pcm: i8) -> u16 {
177        let filtered = if PIEZO_EQ_ENABLED {
178            self.piezo_notch.process(raw_pcm as f32 / 128.0) * 128.0
179        } else {
180            raw_pcm as f32
181        };
182
183        let bit = self.sigma_delta.shape(filtered.clamp(-128.0, 127.0) as i8);
184        if bit > 0 { 1000 } else { 0 }
185    }
186}

188/// Primary Embassy Async Audio Background Task.
189/// Runs alongside your main async application loop without blocking.
190#[task]
191pub async fn embassy_audio_task() {
192    let mut engine = EmbassyAudioEngine::new();
193    engine.start_song();
194
195    loop {
196        // 1. Check for incoming asynchronous Sound Effect triggers (non-blocking poll)
197        if let Ok(cue) = SFX_CHANNEL.try_receive() {
198            match cue {
199                SfxCue::Bootup => { /* trigger FM chime C5 -> E5 -> G5 -> C6 */ }
200                SfxCue::LaserPew => { /* trigger rapid pitch-drop square chirp */ }
201                SfxCue::Shutdown => { /* trigger power-down sweep */ }
202                SfxCue::CoinCollect => { /* trigger 2-tone arpeggio */ }
203                SfxCue::PowerUpFanfare => { /* trigger 4-note fanfare */ }
204                SfxCue::Explosion => { /* trigger 16-bit LFSR noise + sub bass */ }
205                SfxCue::ErrorAlert => { /* trigger 120Hz dual pulse */ }
206                SfxCue::Jump => { /* trigger triangle sweep */ }
207            }
208        }
209
210        // 2. Advance sequencer step
211        if engine.is_playing {
212            engine.current_step = (engine.current_step + 1) % (TOTAL_STEPS as u16);
213        }
214
215        // 3. Non-blocking async sleep until the next step interval
216        Timer::after(Duration::from_micros(STEP_DURATION_MICROS)).await;
217    }
218}
219"#,
220    );
221
222    code
223}
