5/// Generates a standalone no_std Rust module containing the song data, hardware timer/DMA setup, and player logic.
6pub fn generate_rust_song_code(project: &DawProject) -> String {
7    let mut code = String::new();
8    code.push_str("//! Auto-generated by embedded-audio DAW studio.\n");
9    code.push_str(&format!(
10        "//! Target Profile: {:?} | Output Mode: {:?}\n",
11        project.hardware_target.target_mcu, project.hardware_target.output_mode
12    ));
13    code.push_str("//! Architecture: Cortex-M / RISC-V / embedded-audio no_std.\n\n");
14    code.push_str("#![no_std]\n\n");
15    code.push_str("use embedded_audio::prelude::*;\n");
16    code.push_str("use embedded_audio::dsp::BiquadFilter;\n\n");
17
18    code.push_str(&format!(
19        "pub const SONG_TITLE: &str = \"{}\";\n",
20        project.title
21    ));
22    code.push_str(&format!("pub const SONG_BPM: u16 = {};\n", project.bpm));
23    code.push_str(&format!(
24        "pub const SAMPLE_RATE_HZ: u32 = {};\n",
25        project.sample_rate_hz
26    ));
27    code.push_str(&format!(
28        "pub const TOTAL_STEPS: u32 = {};\n",
29        project.total_steps
30    ));
31    code.push_str(&format!(
32        "pub const STEPS_PER_BEAT: u32 = {};\n",
33        project.steps_per_beat
34    ));
35    code.push_str(&format!(
36        "pub const DMA_HALF_BUFFER_SIZE: usize = {};\n\n",
37        project.hardware_target.dma_buffer_size
38    ));
39
40    let samples_per_step =
41        (project.sample_rate_hz as u64 * 60) / (project.bpm as u64 * project.steps_per_beat as u64);
42    code.push_str(&format!(
43        "pub const SAMPLES_PER_STEP: u32 = {};\n\n",
44        samples_per_step
45    ));
46
47    // Generate Note Events table
48    code.push_str("/// Compact note event (step, note, duration, velocity)\n");
49    code.push_str("#[derive(Clone, Copy)]\n");
50    code.push_str("pub struct NoteEvent {\n");
51    code.push_str("    pub step: u16,\n");
52    code.push_str("    pub note: u8,\n");
53    code.push_str("    pub duration_steps: u8,\n");
54    code.push_str("    pub velocity: u8,\n");
55    code.push_str("}\n\n");
56
57    for track in &project.tracks {
58        let safe_name = track.name.replace([' ', '-', '(', ')'], "_").to_lowercase();
59        code.push_str(&format!(
60            "pub const TRACK_{}_NOTES: &[NoteEvent] = &[\n",
61            safe_name.to_uppercase()
62        ));
63        for n in &track.notes {
64            code.push_str(&format!(
65                "    NoteEvent {{ step: {}, note: {}, duration_steps: {}, velocity: {} }},\n",
66                n.step, n.note, n.duration_steps, n.velocity
67            ));
68        }
69        code.push_str("];\n\n");
70    }
71
72    // Piezo Acoustic Compensator configuration
73    let eq = &project.hardware_target.piezo_eq;
74    code.push_str("/// Piezo Transducer DSP Resonance Compensation Filter Parameters\n");
75    code.push_str(&format!(
76        "pub const PIEZO_EQ_ENABLED: bool = {};\n",
77        eq.enabled
78    ));
79    code.push_str(&format!(
80        "pub const PIEZO_RESONANCE_FREQ_HZ: f32 = {:.1};\n",
81        eq.resonance_freq_hz
82    ));
83    code.push_str(&format!(
84        "pub const PIEZO_NOTCH_DEPTH_DB: f32 = {:.1};\n",
85        eq.notch_depth_db
86    ));
87    code.push_str(&format!(
88        "pub const PWM_CARRIER_FREQ_KHZ: u32 = {};\n\n",
89        eq.carrier_freq_khz
90    ));
91
92    // Generate Hardware DMA ISR Driver template
93    let mcu_isr_name = match project.hardware_target.target_mcu {
94        TargetMcu::Stm32U5 => "GPDMA1_Channel0_IRQHandler",
95        TargetMcu::Stm32F4 => "DMA1_Stream5_IRQHandler",
96        TargetMcu::Rp2040 => "DMA_IRQ_0",
97        TargetMcu::Esp32RiscV => "dma_intr_handler",
98        TargetMcu::GenericCortexM0 => "DMA1_Channel1_IRQHandler",
99    };
100
101    code.push_str(&format!(
102        r#"
103/// Hardware Autonomous DMA Double-Buffer Playback Handler.
104/// Place this in your {} interrupt service routine.
105pub struct HardwarePwmAudioDriver {{
106    pub sigma_delta: SigmaDelta,
107    pub piezo_notch: BiquadFilter,
108    pub dma_buffer: [u16; DMA_HALF_BUFFER_SIZE * 2],
109}}
110
111impl HardwarePwmAudioDriver {{
112    pub fn new() -> Self {{
113        Self {{
114            sigma_delta: SigmaDelta::new(),
115            piezo_notch: BiquadFilter::notch(PIEZO_RESONANCE_FREQ_HZ, 4.0, SAMPLE_RATE_HZ as f32),
116            dma_buffer: [0; DMA_HALF_BUFFER_SIZE * 2],
117        }}
118    }}
119
120    /// Fill active half-buffer during DMA Half-Transfer / Full-Transfer interrupt.
121    pub fn fill_half_buffer(&mut self, half_buf: &mut [u16], pcm_samples: &[i8]) {{
122        for (out_slot, &pcm) in half_buf.iter_mut().zip(pcm_samples.iter()) {{
123            // 1. Apply Piezo Acoustic Resonance Notch Filter
124            let filtered = if PIEZO_EQ_ENABLED {{
125                self.piezo_notch.process(pcm as f32 / 128.0) * 128.0
126            }} else {{
127                pcm as f32
128            }};
129
130            // 2. Shape through 1-bit Sigma-Delta / PDM Modulator
131            let bit = self.sigma_delta.shape(filtered.clamp(-128.0, 127.0) as i8);
132
133            // 3. Map to PWM Timer Compare Register (CCR) value
134            *out_slot = if bit > 0 {{ 1000 }} else {{ 0 }};
135        }}
136    }}
137}}
138
139/// Minimal Song Sequencer state machine for embedded audio playback.
140pub struct EmbeddedSongPlayer {{
141    current_sample: u32,
142    current_step: u16,
143    is_playing: bool,
144}}
145
146impl EmbeddedSongPlayer {{
147    pub const fn new() -> Self {{
148        Self {{
149            current_sample: 0,
150            current_step: 0,
151            is_playing: false,
152        }}
153    }}
154
155    pub fn start(&mut self) {{
156        self.current_sample = 0;
157        self.current_step = 0;
158        self.is_playing = true;
159    }}
160
161    pub fn stop(&mut self) {{
162        self.is_playing = false;
163    }}
164
165    /// Advance sequencer by 1 sample. Returns true if a step transition occurred.
166    pub fn tick_sample(&mut self) -> Option<u16> {{
167        if !self.is_playing {{
168            return None;
169        }}
170        self.current_sample += 1;
171        if self.current_sample >= SAMPLES_PER_STEP {{
172            self.current_sample = 0;
173            let step = self.current_step;
174            self.current_step = (self.current_step + 1) % (TOTAL_STEPS as u16);
175            Some(step)
176        }} else {{
177            None
178        }}
179    }}
180}}
181"#,
182        mcu_isr_name
183    ));
184
185    code
186}
