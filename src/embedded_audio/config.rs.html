52/// Crossfade step in Q8 units so a fade completes in `duration_ms`.
53#[inline]
54pub const fn crossfade_step_q8(duration_ms: u16, sample_rate_hz: u32) -> u8 {
55    if duration_ms == 0 || sample_rate_hz == 0 {
56        return 255;
57    }
58    let samples = (duration_ms as u32 * sample_rate_hz) / 1000;
59    if samples == 0 {
60        return 255;
61    }
62    let step = 255u32.div_ceil(samples);
63
64    if step > 255 { 255 } else { step as u8 }
65}
