1//! Minimal mono WAV writer (8-bit unsigned PCM).
2
3extern crate alloc;
4
5use alloc::string::String;
6use alloc::string::ToString;
7use alloc::vec::Vec;

9/// Write mono 8-bit unsigned PCM WAV (`samples`: 0..=255, silence ≈ 128).
10pub fn write_mono_u8(path: &str, sample_rate_hz: u32, samples: &[u8]) -> Result<(), String> {
11    let data = build_wav_u8(sample_rate_hz, samples);
12    std::fs::write(path, &data).map_err(|e| e.to_string())
13}
14
15pub fn build_wav_u8(sample_rate_hz: u32, samples: &[u8]) -> Vec<u8> {
16    let byte_rate = sample_rate_hz;
17
18    let block_align = 1u16;
19    let data_len = samples.len() as u32;
20    let riff_len = 36 + data_len;
21
22    let mut out = Vec::with_capacity(44 + samples.len());
23    out.extend_from_slice(b"RIFF");
24    out.extend_from_slice(&riff_len.to_le_bytes());
25    out.extend_from_slice(b"WAVEfmt ");
26    out.extend_from_slice(&16u32.to_le_bytes()); // fmt chunk size
27    out.extend_from_slice(&1u16.to_le_bytes()); // PCM
28    out.extend_from_slice(&1u16.to_le_bytes()); // mono
29    out.extend_from_slice(&sample_rate_hz.to_le_bytes());
30    out.extend_from_slice(&byte_rate.to_le_bytes());
31    out.extend_from_slice(&block_align.to_le_bytes());
32    out.extend_from_slice(&8u16.to_le_bytes()); // bits per sample
33    out.extend_from_slice(b"data");
34    out.extend_from_slice(&data_len.to_le_bytes());
35    out.extend_from_slice(samples);
36    out
37}

39/// Convert signed 8-bit engine samples to unsigned WAV bytes.
40pub fn pcm_i8_to_u8(samples: &[i8]) -> Vec<u8> {
41    samples.iter().map(|&s| (s as i16 + 128) as u8).collect()
