1//! IMA ADPCM encoder (host / `std` only). Output matches [`crate::decode::AdpcmStream`] layout.
2
3extern crate alloc;

5use alloc::vec::Vec;

7const STEP_TABLE: [i16; 89] = [
8    7, 8, 9, 10, 11, 12, 13, 14, 16, 17, 19, 21, 23, 25, 28, 31, 34, 37, 41, 45, 50, 55, 60, 66,
9    73, 80, 88, 97, 107, 118, 130, 143, 157, 173, 190, 209, 230, 253, 279, 307, 337, 371, 408, 449,
10    494, 544, 598, 658, 724, 796, 876, 963, 1060, 1166, 1282, 1411, 1552, 1707, 1878, 2066, 2272,
11    2499, 2749, 3024, 3327, 3660, 4026, 4428, 4871, 5358, 5894, 6484, 7132, 7845, 8630, 9493,
12    10442, 11487, 12635, 13899, 15289, 16818, 18500, 20350, 22385, 24623, 27086, 29794, 32767,
13];
14
15const INDEX_TABLE: [i8; 16] = [-1, -1, -1, -1, 2, 4, 6, 8, -1, -1, -1, -1, 2, 4, 6, 8];
16
17struct Encoder {
18    predictor: i16,
19    index: u8,
20}
21
22impl Encoder {
23    fn new(predictor: i16, index: u8) -> Self {
24        Self {
25            predictor,
26            index: index.min(88),
27        }
28    }
29
30    fn encode_sample(&mut self, sample: i16) -> u8 {
31        let step = STEP_TABLE[self.index as usize];
32        let diff = sample.wrapping_sub(self.predictor);
33        let sign = if diff < 0 { 8 } else { 0 };
34        let mut delta = diff.abs();
35        let mut nibble = sign;
36
37        if delta >= step {
38            nibble |= 4;
39            delta -= step;
40        }
41        let mut temp = step >> 1;
42        if delta >= temp {
43            nibble |= 2;
44            delta -= temp;
45        }
46        temp >>= 1;
47        if delta >= temp {
48            nibble |= 1;
49        }
50
51        let diffq = step >> 3;
52        let mut est = diffq;
53        if nibble & 4 != 0 {
54            est += step;
55        }
56        if nibble & 2 != 0 {
57            est += step >> 1;
58        }
59        if nibble & 1 != 0 {
60            est += step >> 2;
61        }
62        if sign != 0 {
63            self.predictor = self.predictor.saturating_sub(est);
64        } else {
65            self.predictor = self.predictor.saturating_add(est);
66        }
67
68        let idx = self.index as i32 + INDEX_TABLE[nibble as usize] as i32;
69        self.index = idx.clamp(0, 88) as u8;
70        nibble as u8
71    }
72}

74/// Encode mono i16 PCM into ADPCM payload (4-byte header + packed nibbles).
75pub fn encode_i16(pcm: &[i16]) -> Vec<u8> {
76    let mut enc = Encoder::new(0, 0);
77    let mut out = Vec::with_capacity(4 + pcm.len() / 2);
78    out.extend_from_slice(&enc.predictor.to_le_bytes());
79    out.push(enc.index);
80    out.push(0);
81
82    let mut high = true;
83    let mut byte = 0u8;
84    for &s in pcm {
85        let n = enc.encode_sample(s);
86        if high {
87            byte = n << 4;
88            high = false;
89        } else {
90            out.push(byte | n);
91            high = true;
92        }
93    }
94    if !high {
95        out.push(byte);
96    }
97    out
98}

100/// Encode 8-bit unsigned PCM (0..255, center 128) to ADPCM payload.
101pub fn encode_u8(pcm: &[u8]) -> Vec<u8> {
102    let pcm16: Vec<i16> = pcm.iter().map(|&b| ((b as i16) - 128) << 8).collect();
103    encode_i16(&pcm16)
104}
